% === Part dividers ===
% Shared content lives in parts/part{1,2,3}.tex, included by both masters.
% This master feeds pandoc for the EPUB: the definition is chosen so pandoc
% expands it into an unnumbered H1 (a part heading in the EPUB TOC, styled by
% h1.unnumbered in kdp/kindle.css) with the epigraph as a blockquote beneath.
% The print master (The_Policy_print.tex) carries the full display treatment.
% #1 numeral, #2 title, #3 epigraph, #4 attribution.
\newcommand{\policypart}[4]{%
  \cleardoublepage
  \chapter*{Part #1: #2}%
  \addcontentsline{toc}{part}{Part #1: #2}%
  \markboth{Part #1: #2}{}%
  \begin{quote}\itshape #3\par\vspace{1em}\upshape\textsc{#4}\end{quote}%
  \clearpage}
